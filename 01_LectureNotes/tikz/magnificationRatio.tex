}

	\addplot[smooth, black, very thick]{V(x,0.0)};
	\addplot[smooth, myBlue, very thick]{V(x,0.03)};
	\addplot[smooth, myOrange, very thick]{V(x,0.05)};
	\addplot[smooth, myGreen, very thick]{V(x,0.10)};
	\addplot[smooth, myRed, very thick]{V(x,0.20)};

\end{axis}
